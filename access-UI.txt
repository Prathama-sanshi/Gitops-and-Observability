

# Step 1: get password for argocd 
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d


# Step: 2 - Port forward 
 kubectl port-forward service/argocd-server -n argocd 8080:443
#http://localhost:8080


#Step: 3 Get Prometheus URL
kubectl port-forward svc/kube-prometheus-stack-prometheus 9090:9090 -n observability
http://localhost:9090

#Step : 4 Get Grafana URL
secret:
kubectl get secret --namespace observability -l app.kubernetes.io/component=admin-secret -o jsonpath="{.items[0].data.admin-password}" | base64 --decode ; echo
kubectl --namespace observability port-forward svc/kube-prometheus-stack-grafana 3000:80
